import pandas as pd
def process(data):
    """
    Main processing function
    """
    df = pd.DataFrame(data)
    # Filter high value sales
    df = df[df['amount'] > 1000]
    # Format dates
    df['date'] = pd.to_datetime(df['date'])
    return df.to_dict('records')
